SQL Lab — How I Work With Data

A collection of practical SQL patterns I use for reporting, analysis and data quality. The examples are portfolio demonstrations and can be adapted to MySQL or SQL Server syntax.

analytics_queries.sql 
-- Daily revenue and subscription performance
SELECT
    CAST(transaction_date AS DATE) AS report_date,
    COUNT(DISTINCT transaction_id) AS transactions,
    COUNT(DISTINCT subscriber_id) AS active_subscribers,
    SUM(revenue) AS total_revenue,
    AVG(revenue) AS avg_transaction_value
FROM subscriptions
WHERE transaction_date >= '2026-01-01'
GROUP BY CAST(transaction_date AS DATE)
ORDER BY report_date;
-- Month-over-month revenue movement
WITH monthly AS (
    SELECT
        DATEFROMPARTS(YEAR(transaction_date), MONTH(transaction_date), 1) AS month_start,
        SUM(revenue) AS revenue
    FROM transactions
    GROUP BY DATEFROMPARTS(YEAR(transaction_date), MONTH(transaction_date), 1)
)
SELECT
    month_start,
    revenue,
    LAG(revenue) OVER (ORDER BY month_start) AS previous_month,
    revenue - LAG(revenue) OVER (ORDER BY month_start) AS revenue_change,
    ROUND(
        100.0 * (revenue - LAG(revenue) OVER (ORDER BY month_start))
        / NULLIF(LAG(revenue) OVER (ORDER BY month_start), 0), 2
    ) AS mom_growth_pct
FROM monthly
ORDER BY month_start;
-- Build a reusable partner-performance dataset
WITH partner_metrics AS (
    SELECT
        content_partner,
        COUNT(DISTINCT subscriber_id) AS subscribers,
        SUM(revenue) AS revenue,
        COUNT(*) AS transactions
    FROM content_transactions
    GROUP BY content_partner
),
ranked AS (
    SELECT
        *,
        DENSE_RANK() OVER (ORDER BY revenue DESC) AS revenue_rank
    FROM partner_metrics
)
SELECT *
FROM ranked
WHERE revenue_rank <= 10
ORDER BY revenue_rank;
-- Data-quality checks before reporting
SELECT
    COUNT(*) AS total_rows,
    SUM(CASE WHEN subscriber_id IS NULL THEN 1 ELSE 0 END) AS null_subscribers,
    SUM(CASE WHEN revenue IS NULL THEN 1 ELSE 0 END) AS null_revenue,
    SUM(CASE WHEN revenue < 0 THEN 1 ELSE 0 END) AS negative_revenue,
    COUNT(*) - COUNT(DISTINCT transaction_id) AS duplicate_transactions
FROM transactions;

-- Find duplicate transaction IDs
SELECT transaction_id, COUNT(*) AS duplicate_count
FROM transactions
GROUP BY transaction_id
HAVING COUNT(*) > 1
ORDER BY duplicate_count DESC;
-- Rank content partners within each month
SELECT
    month_start,
    content_partner,
    revenue,
    RANK() OVER (
        PARTITION BY month_start
        ORDER BY revenue DESC
    ) AS monthly_rank,
    SUM(revenue) OVER (
        PARTITION BY month_start
    ) AS monthly_total,
    ROUND(
        100.0 * revenue /
        NULLIF(SUM(revenue) OVER (PARTITION BY month_start), 0), 2
    ) AS revenue_share_pct
FROM partner_monthly_revenue
ORDER BY month_start, monthly_rank;
-- Simple subscriber cohort retention view
WITH first_activity AS (
    SELECT
        subscriber_id,
        MIN(CAST(activity_date AS DATE)) AS first_date
    FROM subscriber_activity
    GROUP BY subscriber_id
)
SELECT
    DATEFROMPARTS(YEAR(f.first_date), MONTH(f.first_date), 1) AS cohort_month,
    DATEFROMPARTS(YEAR(a.activity_date), MONTH(a.activity_date), 1) AS activity_month,
    COUNT(DISTINCT a.subscriber_id) AS active_subscribers
FROM first_activity f
JOIN subscriber_activity a
    ON a.subscriber_id = f.subscriber_id
GROUP BY
    DATEFROMPARTS(YEAR(f.first_date), MONTH(f.first_date), 1),
    DATEFROMPARTS(YEAR(a.activity_date), MONTH(a.activity_date), 1)
ORDER BY cohort_month, activity_month;
